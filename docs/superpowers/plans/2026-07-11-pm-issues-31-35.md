```ts
export function buildAdhocChartOption(series: MetricSeries[], buckets: string[]) {
  return {
    grid: { left: 50, right: 16, top: 58, bottom: 40 },
    xAxis: { type: 'category', data: buckets, name: 'start_time', nameLocation: 'middle', nameGap: 24 },
    yAxis: { type: 'value' },
    series: series.map((s) => ({ name: s.name, type: 'line', smooth: true, showSymbol: true, symbolSize: 4, data: s.values, connectNulls: false })),
    tooltip: { trigger: 'axis' },
    legend: { type: 'scroll', top: 0, left: 8, right: 8 },
  };
}
```
