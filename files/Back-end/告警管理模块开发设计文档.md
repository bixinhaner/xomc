## 一、概述

### 1.1 功能域

告警管理模块 (F04) 负责处理基站设备上报的告警事件，包括告警接收、去重、存储、查询、确认、清除等全生命周期管理。

### 1.2 子模块划分

| 子模块 | 功能说明 |
|--------|---------|
| **告警库** | 告警定义/模板管理，支持国际化，作为告警的标准定义库 |
| **活动告警** | 当前未清除的告警查询与操作 |
| **历史告警** | 已清除告警的查询与归档 |
| **告警过滤** | 告警过滤规则，控制告警入库行为 |
| **告警统计** | 告警数据统计与报表 |

### 1.3 与现有实现的差异分析

| 需求项 | 现有实现 | 需要调整 |
|--------|---------|---------|
| 告警库 | ❌ 无 | ✅ 新增 `alarm_libraries` 表 |
| 活动告警字段 | 部分支持 | ✅ 新增：网元定位、是否已读、明确原因、告警次数、告警类型、描述 |
| 历史告警字段 | 部分支持 | ✅ 新增：清除时间、明确原因、告警次数 |
| 告警过滤 | `alarm_rules` 表 | ✅ 增强/新增 `alarm_filters` 表 |
| 数据权限 | ❌ 无 | ✅ 基于设备组/角色实现 |
| 国际化 | ❌ 无 | ✅ 新增 `alarm_library_i18n` 表 |
| 操作日志 | `audit_logs` 表 | ✅ 复用现有表 |

## 二、数据库设计

### 2.0 时间戳字段维护策略

本项目 **不使用数据库触发器** 维护 `updated_at` 字段，而是采用 **应用层统一处理** 的方式。

**原因：**
1. **可测试性**：应用层逻辑更容易单元测试，不依赖数据库函数
2. **可移植性**：不依赖 PostgreSQL 特定语法，便于迁移其他数据库
3. **可控性**：时间戳更新逻辑在代码中可见，便于调试和维护

**实现方式：**

在 Repository 层的 Update 方法中统一处理：

```go
// internal/alarm/pg_library_repository.go
func (r *PgAlarmLibraryRepository) Update(ctx context.Context, lib *AlarmLibrary) error {
    // 应用层自动设置 updated_at
    lib.UpdatedAt = time.Now()

    query, args, err := psql.Update("alarm_libraries").
        Set("name", lib.Name).
        Set("alarm_code", lib.AlarmCode).
        // ... 其他字段
        Set("updated_at", lib.UpdatedAt).  // 显式设置
        Where(squirrel.Eq{"id": lib.ID}).
        ToSql()
    // ...
}
```

**统一工具函数：**

```go
// internal/core/repository/base.go
package repository

import "time"

// SetUpdatedAt 更新实体的 updated_at 字段
func SetUpdatedAt(entity interface{ SetUpdatedAt(time.Time) }) {
    entity.SetUpdatedAt(time.Now())
}

// BeforeUpdateHook 更新前的钩子，用于设置 updated_at
func BeforeUpdateHook(updatedAt *time.Time) {
    *updatedAt = time.Now()
}
```

### 2.1 告警库表 (alarm_libraries)

告警库是告警的标准定义库，存储各种告警的模板信息。

```sql
-- ============================================
-- 告警库表：存储告警定义/模板
-- ============================================
CREATE TABLE alarm_libraries (
    id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    alarm_code      VARCHAR(128) NOT NULL UNIQUE,     -- 告警标识（唯一）
    alarm_source    VARCHAR(64) NOT NULL,             -- 告警源（如：Device、TR069、SNMP等）
    event_type      VARCHAR(64) NOT NULL,             -- 事件类型
    severity        SMALLINT NOT NULL,                -- 告警级别（1-5）
    enabled         BOOLEAN NOT NULL DEFAULT TRUE,    -- 是否启用

    -- 默认中文（兼容性）
    probable_cause  TEXT NOT NULL,                    -- 可能原因（告警名称）- 默认中文
    explanation     TEXT,                             -- 解释/处理建议 - 默认中文

    -- 扩展信息
    additional_info JSONB DEFAULT '{}',                -- 扩展信息
    carrier         VARCHAR(4),                       -- 运营商（NULL=通用）
    technology      VARCHAR(16),                      -- 制式（NULL=通用）

    created_at      TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at      TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- 索引
CREATE INDEX idx_alarm_libraries_source ON alarm_libraries(alarm_source);
CREATE INDEX idx_alarm_libraries_code ON alarm_libraries(alarm_code);
CREATE INDEX idx_alarm_libraries_severity ON alarm_libraries(severity);
CREATE INDEX idx_alarm_libraries_carrier ON alarm_libraries(carrier) WHERE carrier IS NOT NULL;

-- 注意：updated_at 字段由应用层在 UPDATE 操作时自动维护，不使用数据库触发器
```

### 2.4 历史告警表 (alarms_history) - 增强版

```sql
-- ============================================
-- 历史告警表：存储已清除的告警（TimescaleDB超表）
-- ============================================
DROP TABLE IF EXISTS alarms_history CASCADE;

CREATE TABLE alarms_history (
    time                TIMESTAMPTZ NOT NULL,         -- TimescaleDB时间分区列
    alarm_id            UUID NOT NULL,
    device_id           UUID NOT NULL,
    device_sn           VARCHAR(64) NOT NULL,
    device_name         VARCHAR(128),                  -- 新增
    carrier             VARCHAR(4) NOT NULL,
    technology          VARCHAR(16),                   -- 新增

    -- 告警信息
    alarm_code          VARCHAR(32) NOT NULL,
    alarm_source        VARCHAR(64) NOT NULL,          -- 新增
    event_type          VARCHAR(64),                   -- 新增
    severity            SMALLINT NOT NULL,
    alarm_type          VARCHAR(32),                   -- 新增

    -- 描述信息
    probable_cause      TEXT,
    network_location    TEXT,                          -- 新增：网元定位
    description         TEXT,                          -- 新增：用户描述
    explicit_cause      TEXT,                          -- 新增：明确原因
    additional_info     JSONB DEFAULT '{}',

    -- 状态信息
    status              VARCHAR(16) NOT NULL,          -- active, acknowledged, cleared
    ack_count           INT NOT NULL DEFAULT 0,        -- 新增：告警次数

    -- 时间信息
    raised_at           TIMESTAMPTZ NOT NULL,
    first_raised_at     TIMESTAMPTZ NOT NULL,          -- 新增：首次产生时间
    cleared_at          TIMESTAMPTZ NOT NULL,          -- 清除时间
    acknowledged_at     TIMESTAMPTZ,
    acknowledged_by     VARCHAR(128)
);

-- TimescaleDB超表配置
DO $$
BEGIN
    IF EXISTS (SELECT 1 FROM pg_extension WHERE extname = 'timescaledb') THEN
        PERFORM create_hypertable('alarms_history', 'time',
                                  chunk_time_interval => INTERVAL '7 days',
                                  if_not_exists => TRUE);
        PERFORM add_retention_policy('alarms_history', INTERVAL '365 days', if_not_exists => TRUE);
    END IF;
END $$;

-- 索引
CREATE INDEX idx_alarms_history_device ON alarms_history (device_id, time DESC);
CREATE INDEX idx_alarms_history_alarm ON alarms_history (alarm_id, time DESC);
CREATE INDEX idx_alarms_history_device_sn ON alarms_history (device_sn, time DESC);
CREATE INDEX idx_alarms_history_time ON alarms_history (time DESC);
CREATE INDEX idx_alarms_history_severity ON alarms_history (severity, time DESC);
CREATE INDEX idx_alarms_history_cleared_at ON alarms_history (cleared_at DESC);
```

### 2.5 告警过滤规则表 (alarm_filters)

```sql
-- ============================================
-- 告警过滤规则表
-- ============================================
CREATE TABLE alarm_filters (
    id                  UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name                VARCHAR(255) NOT NULL,         -- 规则名称

    -- 过滤条件
    filter_type         VARCHAR(32) NOT NULL,          -- all, device, device_group
    alarm_sources       VARCHAR(64)[] DEFAULT '{}',    -- 告警源列表（多选）
    alarm_codes         VARCHAR(128)[] DEFAULT '{}',   -- 告警代码列表

    -- 过滤数据类型关联
    device_ids          UUID[] DEFAULT '{}',           -- 关联设备（filter_type=device时）
    device_group_ids    UUID[] DEFAULT '{}',           -- 关联设备组（filter_type=device_group时）

    -- 执行动作
    action              VARCHAR(32) NOT NULL,          -- ignore, auto_acknowledge, auto_clear
    acknowledge_desc    TEXT,                          -- 自动确认时的描述

    -- 排序
    priority            INT NOT NULL DEFAULT 0,        -- 优先级/顺序（1-9），越小越优先

    -- 状态
    enabled             BOOLEAN NOT NULL DEFAULT TRUE,

    -- 审计
    created_by          VARCHAR(128),
    created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_by          VARCHAR(128),
    updated_at          TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    -- 约束：同一告警代码只能有一个生效的过滤规则
    CONSTRAINT unique_alarm_code_filter EXCLUDE (alarm_codes WITH =) WHERE (enabled = TRUE)
);

-- 索引
CREATE INDEX idx_alarm_filters_enabled ON alarm_filters (enabled, priority);
CREATE INDEX idx_alarm_filters_action ON alarm_filters (action);
CREATE INDEX idx_alarm_filters_sources ON alarm_filters USING GIN (alarm_sources);
CREATE INDEX idx_alarm_filters_codes ON alarm_filters USING GIN (alarm_codes);
CREATE INDEX idx_alarm_filters_devices ON alarm_filters USING GIN (device_ids);
CREATE INDEX idx_alarm_filters_groups ON alarm_filters USING GIN (device_group_ids);

-- 注意：updated_at 字段由应用层在 UPDATE 操作时自动维护，不使用数据库触发器
```

### 5.1.1 过滤规则处理逻辑详解

#### ignore（不入库不显示）

```
匹配条件 → 告警代码在过滤规则列表中
执行动作  → 直接丢弃，不进行任何数据库操作
```

**代码示例：**

```go
case FilterActionIgnore:
    // 记录过滤日志（可选，用于审计）
    e.logger.Debug("alarm filtered by rule",
        zap.String("alarm_code", alarm.AlarmCode),
        zap.String("filter_name", filter.Name))
    // 不入库，直接返回
    return nil, nil
```

#### auto_acknowledge（自动确认）

```
匹配条件 → 告警代码在过滤规则列表中
执行动作  →
    1. 去重检查 (device_sn + alarm_code)
    2. 新增：插入活动告警表，status=acknowledged
    3. 已存在：更新 count++，更新时间，保持 status=acknowledged
    4. 发送告警事件（含自动确认标记）
```

**代码示例：**

```go
case FilterActionAutoAcknowledge:
    // 去重检查
    existing, err := e.store.GetActiveByDeviceAndCode(ctx, alarm.DeviceSN, alarm.AlarmCode)
    if err == nil {
        // 已存在，更新
        existing.AckCount++
        existing.LastUpdatedAt = time.Now()
        existing.Status = model.AlarmStatusAcknowledged
        existing.Description = filter.AcknowledgeDesc  // 自动确认描述
        existing.AcknowledgedAt = &now
        existing.AcknowledgedBy = "system:auto_filter"
        return e.store.UpdateActive(ctx, existing)
    }

    // 不存在，插入新告警（状态为已确认）
    alarm.Status = model.AlarmStatusAcknowledged
    alarm.AckCount = 1
    alarm.Description = filter.AcknowledgeDesc
    alarm.AcknowledgedAt = &now
    alarm.AcknowledgedBy = "system:auto_filter"
    return e.store.SaveActive(ctx, alarm)
```

#### auto_clear（自动清除）

```
匹配条件 → 告警代码在过滤规则列表中
执行动作  →
    1. 去重检查
    2. 直接写入历史告警表，status=cleared
    3. cleared_at = 当前时间
    4. 可选：发送清除事件
```

**代码示例：**

```go
case FilterActionAutoClear:
    now := time.Now()

    // 去重检查（历史表也需要去重）
    existing, err := e.store.GetHistoryByDeviceAndCode(ctx, alarm.DeviceSN, alarm.AlarmCode)
    if err == nil && existing != nil {
        // 历史表中已存在，更新清除时间
        existing.ClearedAt = &now
        return e.store.UpdateHistory(ctx, existing)
    }

    // 活动表中存在，先从活动表移除
    active, err := e.store.GetActiveByDeviceAndCode(ctx, alarm.DeviceSN, alarm.AlarmCode)
    if err == nil && active != nil {
        _ = e.store.RemoveActive(ctx, active.ID)
    }

    // 写入历史表（状态为已清除）
    alarm.Status = model.AlarmStatusCleared
    alarm.ClearedAt = &now
    alarm.AcknowledgedAt = &now
    alarm.AcknowledgedBy = "system:auto_filter"
    return e.store.Archive(ctx, alarm)
```

#### 无匹配规则（默认流程）

```
匹配条件 → 告警代码不在任何过滤规则中
执行动作  →
    1. 告警库匹配，补充告警信息
    2. 去重检查
    3. 新增：插入活动告警表，status=active
    4. 已存在：更新 count++，更新时间
    5. 发送告警事件
```

### 5.3 告警过滤引擎实现

```go
// internal/alarm/filter_engine.go
package alarm

import (
	"context"
	"fmt"
	"time"

	"github.com/google/uuid"
	"github.com/omcgo/omcgo/internal/core/model"
	"go.uber.org/zap"
)

// FilterEngine 告警过滤引擎
type FilterEngine struct {
	filterRepo AlarmFilterRepository
	store      AlarmStore
	libraryRepo AlarmLibraryRepository
	logger     *zap.Logger
}

// ProcessAlarm 处理告警（带过滤规则）
func (e *FilterEngine) ProcessAlarm(ctx context.Context, alarm *model.Alarm) error {
	// 1. 尝试匹配过滤规则
	result, err := e.Match(ctx, alarm)
	if err != nil {
		return fmt.Errorf("filter match failed: %w", err)
	}

	now := time.Now()

	// 2. 根据匹配结果处理
	if result.Matched {
		return e.executeFilterAction(ctx, alarm, result, now)
	}

	// 3. 无匹配规则，走默认流程
	return e.processDefault(ctx, alarm, now)
}

// executeFilterAction 执行过滤动作
func (e *FilterEngine) executeFilterAction(
	ctx context.Context,
	alarm *model.Alarm,
	result *MatchResult,
	now time.Time,
) error {
	switch result.Action {
	case FilterActionIgnore:
		// 不入库，记录日志后直接返回
		e.logger.Debug("alarm ignored by filter",
			zap.String("alarm_code", alarm.AlarmCode),
			zap.String("device_sn", alarm.DeviceSN),
			zap.String("filter_name", result.FilterName))
		return nil

	case FilterActionAutoAcknowledge:
		return e.processAutoAcknowledge(ctx, alarm, result, now)

	case FilterActionAutoClear:
		return e.processAutoClear(ctx, alarm, result, now)

	default:
		return fmt.Errorf("unknown filter action: %s", result.Action)
	}
}

// processAutoAcknowledge 处理自动确认动作
func (e *FilterEngine) processAutoAcknowledge(
	ctx context.Context,
	alarm *model.Alarm,
	result *MatchResult,
	now time.Time,
) error {
	// 去重检查
	existing, err := e.store.GetActiveByDeviceAndCode(ctx, alarm.DeviceSN, alarm.AlarmCode)
	if err == nil && existing != nil {
		// 已存在，更新
		existing.AckCount++
		existing.LastUpdatedAt = now
		existing.Status = model.AlarmStatusAcknowledged
		existing.Description = result.AckDesc
		existing.AcknowledgedAt = &now
		existing.AcknowledgedBy = "system:auto_filter:" + result.FilterName
		return e.store.UpdateActive(ctx, existing)
	}

	// 不存在，插入新告警（状态为已确认）
	alarm.Status = model.AlarmStatusAcknowledged
	alarm.AckCount = 1
	alarm.Description = result.AckDesc
	alarm.AcknowledgedAt = &now
	alarm.AcknowledgedBy = "system:auto_filter:" + result.FilterName
	alarm.CreatedAt = now
	alarm.UpdatedAt = now
	alarm.FirstRaisedAt = now
	alarm.LastUpdatedAt = now

	// 从告警库补充信息
	e.enrichFromLibrary(ctx, alarm)

	return e.store.SaveActive(ctx, alarm)
}

// processAutoClear 处理自动清除动作
func (e *FilterEngine) processAutoClear(
	ctx context.Context,
	alarm *model.Alarm,
	result *MatchResult,
	now time.Time,
) error {
	// 先检查活动表，存在则移除
	active, err := e.store.GetActiveByDeviceAndCode(ctx, alarm.DeviceSN, alarm.AlarmCode)
	if err == nil && active != nil {
		_ = e.store.RemoveActive(ctx, active.ID)
	}

	// 写入历史表（状态为已清除）
	alarm.Status = model.AlarmStatusCleared
	alarm.ClearedAt = &now
	alarm.AcknowledgedAt = &now
	alarm.AcknowledgedBy = "system:auto_filter:" + result.FilterName
	alarm.CreatedAt = now
	alarm.FirstRaisedAt = now
	alarm.RaisedAt = now

	// 从告警库补充信息
	e.enrichFromLibrary(ctx, alarm)

	return e.store.Archive(ctx, alarm)
}

// processDefault 处理无匹配规则的默认流程
func (e *FilterEngine) processDefault(
	ctx context.Context,
	alarm *model.Alarm,
	now time.Time,
) error {
	// 去重检查
	existing, err := e.store.GetActiveByDeviceAndCode(ctx, alarm.DeviceSN, alarm.AlarmCode)
	if err == nil && existing != nil {
		// 已存在，更新
		existing.AckCount++
		existing.LastUpdatedAt = now
		existing.RaisedAt = now
		return e.store.UpdateActive(ctx, existing)
	}

	// 不存在，插入新告警（状态为活动）
	alarm.Status = model.AlarmStatusActive
	alarm.AckCount = 1
	alarm.CreatedAt = now
	alarm.UpdatedAt = now
	alarm.FirstRaisedAt = now
	alarm.LastUpdatedAt = now
	alarm.RaisedAt = now

	// 从告警库补充信息
	e.enrichFromLibrary(ctx, alarm)

	return e.store.SaveActive(ctx, alarm)
}

// enrichFromLibrary 从告警库补充告警信息
func (e *FilterEngine) enrichFromLibrary(ctx context.Context, alarm *model.Alarm) {
	lib, err := e.libraryRepo.GetByCode(ctx, alarm.AlarmCode)
	if err == nil && lib != nil {
		// 补充告警库中的信息
		if alarm.ProbableCause == "" {
			alarm.ProbableCause = lib.ProbableCause
		}
		alarm.AlarmSource = lib.AlarmSource
		alarm.EventType = lib.EventType
		alarm.Severity = lib.Severity
	}
}

// Match 匹配过滤规则
func (e *FilterEngine) Match(ctx context.Context, alarm *model.Alarm) (*MatchResult, error) {
	// 获取所有启用的过滤规则，按priority升序
	filters, err := e.filterRepo.ListEnabled(ctx)
	if err != nil {
		return nil, err
	}

	// 按优先级顺序匹配
	for _, filter := range filters {
		if e.matchFilter(alarm, filter) {
			return &MatchResult{
				Matched:    true,
				Action:     filter.Action,
				AckDesc:    filter.AcknowledgeDesc,
				FilterID:   filter.ID,
				FilterName: filter.Name,
			}, nil
		}
	}

	return &MatchResult{Matched: false}, nil
}

// matchFilter 判断告警是否匹配过滤规则
func (e *FilterEngine) matchFilter(alarm *model.Alarm, filter *AlarmFilter) bool {
	// 1. 检查告警源
	if len(filter.AlarmSources) > 0 && !contains(filter.AlarmSources, alarm.AlarmSource) {
		return false
	}

	// 2. 检查告警代码
	if len(filter.AlarmCodes) > 0 && !contains(filter.AlarmCodes, alarm.AlarmCode) {
		return false
	}

	// 3. 根据过滤类型检查
	switch filter.FilterType {
	case FilterTypeAll:
		return true

	case FilterTypeDevice:
		if len(filter.DeviceIDs) > 0 {
			for _, deviceID := range filter.DeviceIDs {
				if deviceID == alarm.DeviceID {
					return true
				}
			}
			return false
		}
		return true

	case FilterTypeDeviceGroup:
		// 需要查询设备所属的设备组
		groups, err := e.filterRepo.GetDeviceGroups(ctx, alarm.DeviceID)
		if err != nil {
			return false
		}
		for _, groupID := range filter.DeviceGroupIDs {
			if contains(groups, groupID) {
				return true
			}
		}
		return false
	}

	return false
}

// contains 辅助函数
func contains(slice []string, item string) bool {
	for _, s := range slice {
		if s == item {
			return true
		}
	}
	return false
}
```

```go
	Technologies []string         `json:"technologies,omitempty"`  // 网络类型白名单

// DataPermissionChecker 数据权限检查器
type DataPermissionChecker struct {
	userRepo  admin.UserRepository
	groupRepo topology.DeviceGroupRepository
	// 可选：设备仓储，用于通过网络类型查询设备
	deviceRepo DeviceRepository
}

// ApplyDataPermission 应用数据权限到查询过滤器
func (c *DataPermissionChecker) ApplyDataPermission(
	ctx context.Context,
	filter *AlarmQueryFilter,
	userID uuid.UUID,
) error {
	// 1. 获取用户信息
	user, err := c.userRepo.GetByID(ctx, userID)
	if err != nil {
		return err
	}

	// 2. 检查是否是超级管理员（查看所有权限）
	if c.isSuperAdmin(user) {
		return nil  // 不限制
	}

	// 3. 获取用户的角色及其数据权限
	roles, err := c.userRepo.GetRoles(ctx, userID)
	if err != nil {
		return err
	}

	// 4. 收集用户可访问的设备ID（基于设备组）
	accessibleDeviceIDs, err := c.getAccessibleDeviceIDs(ctx, roles)
	if err != nil {
		return err
	}

	// 5. 收集用户可访问的网络类型
	accessibleTechnologies := c.getAccessibleTechnologies(roles)

	// 6. 应用过滤条件
	filter.DeviceIDs = accessibleDeviceIDs
	filter.Technologies = accessibleTechnologies

	// 7. 如果用户绑定了运营商，也应用运营商过滤
	if user.Carrier != nil {
		filter.Carrier = user.Carrier
	}

	return nil
}

// isSuperAdmin 判断是否是超级管理员
func (c *DataPermissionChecker) isSuperAdmin(user *admin.User) bool {
	for _, role := range user.Roles {
		if role.Name == "admin" || role.Name == "super_admin" {
			return true
		}
	}
	return false
}

// getAccessibleDeviceIDs 获取用户可访问的设备ID列表（基于设备组）
func (c *DataPermissionChecker) getAccessibleDeviceIDs(
	ctx context.Context,
	roles []admin.Role,
) ([]uuid.UUID, error) {
	var deviceIDs []uuid.UUID

	for _, role := range roles {
		// 角色中的 DeviceGroups 字段存储可访问的设备组
		if len(role.DeviceGroups) > 0 {
			for _, groupID := range role.DeviceGroups {
				members, err := c.groupRepo.ListDeviceIDs(ctx, groupID)
				if err != nil {
					continue
				}
				deviceIDs = append(deviceIDs, members...)
			}
		}
	}

	return uniqueDeviceIDs(deviceIDs), nil
}

// getAccessibleTechnologies 获取用户可访问的网络类型列表
func (c *DataPermissionChecker) getAccessibleTechnologies(roles []admin.Role) []string {
	techSet := make(map[string]bool)

	for _, role := range roles {
		// 角色中的 Technologies 字段存储可访问的网络类型
		for _, tech := range role.Technologies {
			techSet[tech] = true
		}
	}

	// 转换为切片
	result := make([]string, 0, len(techSet))
	for tech := range techSet {
		result = append(result, tech)
	}
	return result
}

// uniqueDeviceIDs 去重设备ID
func uniqueDeviceIDs(ids []uuid.UUID) []uuid.UUID {
	seen := make(map[uuid.UUID]bool)
	result := make([]uuid.UUID, 0, len(ids))

	for _, id := range ids {
		if !seen[id] {
			seen[id] = true
			result = append(result, id)
		}
	}
	return result
}
```
